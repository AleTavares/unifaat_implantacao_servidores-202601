#!/bin/bash
END=$((SECONDS+300))
COUNT=0
while [ $SECONDS -lt $END ]; do
    curl -s -o /dev/null "http://$ENDPOINT"
    COUNT=$((COUNT+1))
done
echo "Total de requisicoes em 5 min: $COUNT"
EOF
chmod +x load-test.sh
